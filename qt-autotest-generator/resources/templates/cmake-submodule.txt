cmake_minimum_required(VERSION 3.16)

# Qt 版本（由 test_writer 从 session.qt_version 填入）
set(QT_VERSION "{QT_VERSION}")  # 5 或 6

# 项目库名称（由 test_writer 从 dependency_tracer 的 source_dirs 推断填入）
set(PROJECT_LIBRARIES "{PROJECT_LIBRARIES}")

# {QT_EXTRA_LIBS} 由 test_writer 填入：GUI 类填 "Qt${QT_VERSION}::Widgets"，
# 纯 Core 模块填空字符串。避免无条件链接 Widgets 导致 Core-only 项目编译失败。

# 代码覆盖率：Debug 模式下启用 gcov 插桩
# run-ut.sh 的 step_5 (lcov --capture) 依赖此标志采集覆盖率数据
# report_generator 从 build-{test_dir}/coverage/filtered.info 解析覆盖率
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fprofile-arcs -ftest-coverage -O0")
    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -lgcov")
    message(STATUS "UT: Code coverage enabled for test_{module_name}")
endif()

# 收集所有测试文件
file(GLOB TEST_SOURCES "test_*.cpp")

# stub-ext 运行时依赖
set(STUB_SHADOW "${CMAKE_SOURCE_DIR}/{test_dir}/3rdparty/stub/stub-shadow.cpp")

if(NOT TEST_SOURCES)
    message(WARNING "No test files found in ${CMAKE_CURRENT_SOURCE_DIR}")
    return()
endif()

# 创建测试可执行文件
add_executable(test_{module_name} ${TEST_SOURCES} ${STUB_SHADOW})

# 链接依赖
target_link_libraries(test_{module_name}
    PRIVATE
    GTest::gtest
    GTest::gtest_main
    Qt${QT_VERSION}::Core
    {QT_EXTRA_LIBS}
)

# 如果项目有自定义库，添加
if(PROJECT_LIBRARIES AND NOT PROJECT_LIBRARIES STREQUAL "")
    target_link_libraries(test_{module_name}
        PRIVATE
        ${PROJECT_LIBRARIES}
    )
endif()

# 覆盖率库（Debug 模式下）
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_link_libraries(test_{module_name} PRIVATE gcov)
endif()

# 包含目录
target_include_directories(test_{module_name}
    PRIVATE
    ${CMAKE_SOURCE_DIR}/{test_dir}/3rdparty/stub
    ${CMAKE_SOURCE_DIR}/{source_module_path}
)

# 自动发现测试
gtest_discover_tests(test_{module_name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

message(STATUS "UT: test_{module_name} configured with ${TEST_SOURCES}")
